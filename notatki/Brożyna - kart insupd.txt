    b. Wszystkim klientom z Opatowa zmianić opiekuna na Martę Havel

SELECT  customer.city       AS 'Miasto',
	customer.contact_id AS 'ID opiekuna',
        emp.last_name       AS 'Nazwisko pracownika',
        emp.first_name      AS 'Imię pracownika'
 FROM   customer INNER JOIN emp ON customer.contact_id = emp.id
WHERE   customer.city = 'Opatów';

UPDATE customer
   SET customer.contact_id =   (SELECT emp.id
				  FROM emp
				 WHERE emp.first_name = 'Marta' AND emp.last_name = 'Havel')
 WHERE customer.city = 'Opatów';
