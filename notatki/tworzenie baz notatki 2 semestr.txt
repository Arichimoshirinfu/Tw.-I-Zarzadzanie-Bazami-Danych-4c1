A. Wyświetl ID, nazwisko i imię pracowników "Jan Nowak" urodzony w 1999 roku i "Adam Kowalski" urodzony w 1997 roku.

SELECT  id 		AS 'ID',
	last_name 	AS 'Nazwisko',
	first_name  	AS 'Imie'
 FROM 	emp
WHERE   (last_name = 'Nowak' AND first_name = 'Jan' AND year(birth_date)=1999)
   OR   (last_name = 'Kowalski' AND first_name = 'Adam' AND year(birth_date)=1997)

B. Podaj id, miasto i kraj dla wszystkich magazynów z Europy

SELECT 	warehouse.id 		AS 'ID',
	warehouse.city 		AS 'Miasto',
 	warehouse.country	AS 'Kraj'
  FROM  warehouse INNER JOIN region ON warehouse.region_id = region.id
 WHERE  region.name LIKE '%Europe%';

SELECT  customer.id 	AS 'ID',
	customer.name	AS 'nazwa',
	customer.city	AS 'miasto',
	region.name	AS 'nazwa regionu',
	emp.last_name 	AS 'Nazwisko opiekuna'
 FROM 	emp      INNER JOIN customer 	   ON emp.id = customer.contact_id
		 INNER JOIN region	   ON region.id = customer.region_id
WHERE customer.city LIKE '%Kielce%' AND customer.name LIKE 'N%';
