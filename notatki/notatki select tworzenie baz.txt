SELECT  customer.id 		AS 'ID',
        customer.name       AS 'Nazw klienta',
        customer.city		AS 'Miasto klienta',
        region.name			AS 'region klienta',
        emp.last_name		AS 'Nazw opiekuna'
FROM	customer INNER JOIN region ON region.id = customer.region_id
				 INNER JOIN emp    ON emp.id = customer.contact_id
WHERE   customer.last_name LIKE 'N%' OR customer.last_name LIKE 'P%';
